# P1a — Scout (昨日复盘, full depth) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Scout for the single 昨日复盘 (prior-session recap) dimension at full depth — fetch real domestic sources, extract `Source`+`Quote` records with exact, frozen spans, trace one media number to its primary disclosure, persist idempotently, and expose a grounded-floor evaluation — all unit-testable offline via fixtures.

**Architecture:** Pure, offline-testable core (value objects, Chinese `publish_ts` parsing, financial-number detection, main-content cleaning, span extraction, floor evaluation) sits behind thin live-I/O clients (akshare, Bocha, Playwright). Fixtures-first (D-P1-4): every parse/extract/floor path is exercised on saved HTML/JSON with no network; live clients are exercised only in a manual smoke run. Persistence is upsert-by-canonical-url so repeated runs against the same prior session don't duplicate the pool.

**Tech Stack:** Python 3.11+, asyncpg (raw SQL, no ORM), Postgres 16, akshare (market numbers + trade calendar), httpx (Bocha), Playwright (Chromium full-text fetch), trafilatura (pure-Python Chinese main-content extraction), pytest + pytest-asyncio.

**Reference docs:** `docs/superpowers/specs/2026-06-01-p1-walking-skeleton-design.md` (the spec — authority), `CITATION_CONTRACT.md` §3.2/§3.3 (Source/Quote), `citation/glossary.md` (primary / independent), `docs/scout_claude_code_prompt.md` (full-grade brief — P1 is a deliberate subset), `DECISIONS.md` (D-22 calendar cache, D-28 domestic-only).

**Workflow rule (from the user):** implement a task → run its tests and read the output → only then advance. Temp artifacts (fetched HTML, akshare snapshots) may be saved to disk under `tests/scout/fixtures/`. Commit after each task.

`marketwind/scout/timeparse.py`:

```python
"""Parse Chinese / relative publish-time strings to tz-aware datetimes.

Year-boundary rule (spec §6): a bare 月日 with no year resolves to the most
recent occurrence at or before ingest time; relative terms resolve against
ingested_at in its tz. This gates I9, so a wrong year silently drops sources.
"""

from __future__ import annotations

import re
from datetime import datetime, timedelta, timezone

CN_TZ = timezone(timedelta(hours=8))  # Asia/Shanghai, no DST

_FULL = re.compile(r"(?P<y>\d{4})年(?P<m>\d{1,2})月(?P<d>\d{1,2})日(?:\s*(?P<H>\d{1,2}):(?P<M>\d{2}))?")
_MD = re.compile(r"(?P<m>\d{1,2})月(?P<d>\d{1,2})日(?:\s*(?P<H>\d{1,2}):(?P<M>\d{2}))?")
_HOURS_AGO = re.compile(r"(?P<n>\d+)\s*小时前")
_MIN_AGO = re.compile(r"(?P<n>\d+)\s*分钟前")


def parse_publish_ts(raw: str, ingested_at: datetime) -> datetime:
    raw = raw.strip()

    m = _FULL.search(raw)
    if m:
        return datetime(
            int(m["y"]), int(m["m"]), int(m["d"]),
            int(m["H"] or 0), int(m["M"] or 0), tzinfo=CN_TZ,
        )

    m = _HOURS_AGO.search(raw)
    if m:
        return ingested_at - timedelta(hours=int(m["n"]))

    m = _MIN_AGO.search(raw)
    if m:
        return ingested_at - timedelta(minutes=int(m["n"]))

    if "今晨" in raw:
        return ingested_at.replace(hour=6, minute=0, second=0, microsecond=0)
    if "今天" in raw or "今日" in raw:
        return ingested_at.replace(hour=0, minute=0, second=0, microsecond=0)

    m = _MD.search(raw)
    if m:
        candidate = datetime(
            ingested_at.year, int(m["m"]), int(m["d"]),
            int(m["H"] or 0), int(m["M"] or 0), tzinfo=CN_TZ,
        )
        if candidate > ingested_at:  # future → most recent occurrence is last year
            candidate = candidate.replace(year=ingested_at.year - 1)
        return candidate

    raise ValueError(f"unparseable publish_ts: {raw!r}")
```

"""Detect whether a Chinese sentence carries a financial quantity.

Conservative: a sentence is quantitative if it contains a digit adjacent to a
financial unit / percentage / magnitude word, or a 涨/跌 + 数字 movement.
"""

from __future__ import annotations

import re

# digit (incl. decimals/commas) followed by a financial unit or magnitude
_NUM_UNIT = re.compile(
    r"\d[\d,\.]*\s*"
    r"(%|％|个百分点|万亿|亿元?|万元?|亿|万|元|股|手|倍|个交易日|bp|BP)"
)
# 涨/跌/增长/下降 + (超)? + 数字
_MOVE = re.compile(r"(涨|跌|增长|下降|下滑|上涨|下跌)\s*超?\s*\d")


def is_quantitative(text: str) -> bool:
    return bool(_NUM_UNIT.search(text) or _MOVE.search(text))
```

"""Derive a normalized domain from a URL. P1: distinct-domain identity only.

Outlet-group normalization (collapsing 新浪系 / 财联社系 to one identity) is a
P2 concern and is intentionally NOT done here (spec §2 Out-of-scope).
"""

from __future__ import annotations

from urllib.parse import urlparse


def domain_of(url: str) -> str:
    host = (urlparse(url).hostname or "").lower()
    if host.startswith("www."):
        host = host[len("www.") :]
    return host
```

`marketwind/scout/clean.py`:

```python
"""Clean a page to frozen main-content text, then extract sentence quotes.

Span-freeze rule (spec §6): clean once, freeze the string as raw_content, and
index every span into THAT string. No re-clean / re-normalize downstream — any
mutation drifts offsets and produces phantom RF-9 failures.
"""

from __future__ import annotations

import re

import trafilatura

from marketwind.scout.domain import Quote
from marketwind.scout.numbers import is_quantitative

# Chinese sentence terminators (keep the terminator with the sentence).
_SENT = re.compile(r"[^。！？]*[。！？]")


def clean_main_content(html: str) -> str:
    """Extract main-article text. Deterministic; the result is the frozen archive."""
    out = trafilatura.extract(
        html,
        favor_recall=True,
        include_comments=False,
        include_tables=False,
        output_format="txt",
    )
    if not out:
        raise ValueError("main-content extraction returned empty")
    # Normalize newlines to single spaces ONCE, here, before freezing.
    return re.sub(r"\s+", " ", out).strip()


def _split_sentences(raw: str) -> list[tuple[int, int, str]]:
    spans = []
    for m in _SENT.finditer(raw):
        text = m.group().strip()
        if not text:
            continue
        start = raw.index(text, m.start())
        spans.append((start, start + len(text), text))
    return spans


def extract_quotes(raw: str) -> list[Quote]:
    sents = _split_sentences(raw)
    quotes: list[Quote] = []
    for i, (start, end, text) in enumerate(sents):
        before = sents[i - 1][2] if i > 0 else None
        after = sents[i + 1][2] if i + 1 < len(sents) else None
        quant = is_quantitative(text)
        quotes.append(
            Quote(
                text=text,
                span_start=start,
                span_end=end,
                context_before=before,
                context_after=after,
                is_quantitative=quant,
                quote_type="data" if quant else "factual",
            )
        )
    return quotes
```

from __future__ import annotations

import asyncpg

from marketwind.scout.domain import Source


async def upsert_source(conn: asyncpg.Connection, source: Source) -> str:
    """Insert or replace a Source by URL and (re)write its quotes. Returns source_id."""
    async with conn.transaction():
        source_id = await conn.fetchval(
            """
            insert into sources (url, domain, byline, publish_ts, is_primary, raw_content)
            values ($1, $2, $3, $4, $5, $6)
            on conflict (url) do update set
                domain = excluded.domain,
                byline = excluded.byline,
                publish_ts = excluded.publish_ts,
                is_primary = excluded.is_primary,
                raw_content = excluded.raw_content,
                ingested_at = now()
            returning source_id;
            """,
            source.url, source.domain, source.byline,
            source.publish_ts, source.is_primary, source.raw_content,
        )
        # Replace quotes wholesale (the source's raw_content may have changed).
        await conn.execute("delete from quotes where source_id = $1;", source_id)
        for q in source.quotes:
            await conn.execute(
                """
                insert into quotes (
                    source_id, text, span_start, span_end,
                    context_before, context_after, quote_type,
                    is_quantitative, attached_primary_url
                ) values ($1,$2,$3,$4,$5,$6,$7,$8,$9);
                """,
                source_id, q.text, q.span_start, q.span_end,
                q.context_before, q.context_after, q.quote_type,
                q.is_quantitative, q.attached_primary_url,
            )
        return str(source_id)

from __future__ import annotations

from datetime import date
from functools import lru_cache


def prev_trading_session(today: date, sessions: list[date]) -> date:
    """Latest trading session strictly before `today`. Never naive yesterday."""
    prior = [d for d in sessions if d < today]
    if not prior:
        raise ValueError(f"no trading session before {today}")
    return max(prior)


@lru_cache(maxsize=1)

    import akshare as ak

    return tuple(d if isinstance(d, date) else d.date() for d in df["trade_date"])


def live_prev_trading_session(today: date) -> date:  # pragma: no cover - live
    return prev_trading_session(today, list(_load_sessions()))

"""Playwright full-text fetch: resource-blocked, layered wait, 15s ceiling.

Pure disposition logic is unit-tested. The live navigation (fetch_html) is
exercised in the smoke run only.
"""

from __future__ import annotations

import enum

_CEILING_MS = 15_000
_BLOCK_TYPES = {"image", "media", "font", "stylesheet"}


class FetchOutcome(enum.Enum):
    TRANSIENT = "transient"  # retry with backoff+jitter
    PERMANENT = "permanent"  # fail fast, don't burn a retry


class PermanentFetchError(Exception):
    pass


def is_permanent_status(status: int) -> bool:
    """404/403/410/451 are permanent; 5xx and timeouts are transient."""
    return status in {403, 404, 410, 451}


def classify_ceiling_hit(content_loaded: bool) -> FetchOutcome:
    """At the 15s ceiling: usable content → transient; nothing → permanent."""
    return FetchOutcome.TRANSIENT if content_loaded else FetchOutcome.PERMANENT


async def fetch_html(context, url: str) -> str:  # pragma: no cover - live navigation
    """Fetch a page's HTML with resource blocking and a layered wait.

    `context` is a reused Playwright BrowserContext (no warm pool in P1).
    """
    page = await context.new_page()
    await page.route(
        "**/*",
        lambda route: (
            route.abort()
            if route.request.resource_type in _BLOCK_TYPES
            else route.continue_()
        ),
    )
    try:
        resp = await page.goto(url, wait_until="domcontentloaded", timeout=_CEILING_MS)
        if resp is not None and is_permanent_status(resp.status):
            raise PermanentFetchError(f"{resp.status} for {url}")
        # Layered wait: try network-idle, fall back to DOM-stability, capped at ceiling.
        try:
            await page.wait_for_load_state("networkidle", timeout=_CEILING_MS)
        except Exception:
            pass  # tickers may never idle; DOM is already loaded → proceed
        return await page.content()
    finally:
        await page.close()
```

"""Scout runner: assemble the pool (live) and evaluate the grounded floor (pure).

Grounded floor (spec §8, provisional 8/1/2 — revisit after first smoke run):
publish only if ALL hold, else the caller emits an honest-shortfall status.
"""

from __future__ import annotations

from marketwind.scout.domain import FloorResult, Source

MIN_GROUNDED_QUOTES = 8
MIN_PRIMARY_QUOTES = 1
MIN_INDEPENDENT_OUTLETS = 2


def evaluate_floor(sources: list[Source]) -> FloorResult:
    grounded = sum(len(s.quotes) for s in sources)
    primary = sum(
        1 for s in sources for q in s.quotes if q.attached_primary_url
    )
    outlets = len({s.domain for s in sources})
    passed = (
        grounded >= MIN_GROUNDED_QUOTES
        and primary >= MIN_PRIMARY_QUOTES
        and outlets >= MIN_INDEPENDENT_OUTLETS
    )
    return FloorResult(
        grounded_quotes=grounded,
        primary_traced_quotes=primary,
        independent_outlets=outlets,
        passed=passed,
    )
```
